    43→function generateCode(): string {
    44→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
    45→  let code = "AP";
    46→  for (let i = 0; i < 8; i++) code += chars.charAt(Math.floor(Math.random() * chars.length));
    47→  return code;
    48→}
