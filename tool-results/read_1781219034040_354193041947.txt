    72→
    73→export default function AppleNetApp() {
    74→  const { t, isRTL, lang } = useLanguage();
    75→
    76→  const NAV_TABS = [
    77→    { id: "home", icon: Home, label: t("nav.home") },
    78→    { id: "cards", icon: Wifi, label: t("nav.cards") },
    79→    { id: "starlink", icon: Satellite, label: "Starlink" },
    80→    { id: "chat", icon: MessageCircle, label: t("nav.chat") },
    81→    { id: "credit", icon: Wallet, label: t("nav.credit") },
    82→  ];
    83→
    84→  const [user, setUser] = useState<FirebaseUser | null>(null);
    85→  const [isAdmin, setIsAdmin] = useState(false);
    86→  const [userRole, setUserRole] = useState<string>("user");
    87→  const [managedNetwork, setManagedNetwork] = useState<string>("");
    88→  const [activeTab, setActiveTab] = useState("home");
    89→  const [showAuth, setShowAuth] = useState(false);
    90→  const [authMode, setAuthMode] = useState<"login" | "register">("login");
    91→  const [showAdmin, setShowAdmin] = useState(false);
    92→  const [loading, setLoading] = useState(true);
    93→  const [showMenu, setShowMenu] = useState(false);
    94→  const [splashDone, setSplashDone] = useState(false);
    95→  const [splashExiting, setSplashExiting] = useState(false);
    96→  const [userName, setUserName] = useState<string>("");
    97→  const [showOnboarding, setShowOnboarding] = useState(false);
    98→  const [splashProgress, setSplashProgress] = useState(0);
    99→  const [showLocationModal, setShowLocationModal] = useState(false);
   100→  const [locationModalProvinceId, setLocationModalProvinceId] = useState("");
   101→  const [locationModalDistrict, setLocationModalDistrict] = useState("");
   102→  const [locationModalSaving, setLocationModalSaving] = useState(false);
   103→  const [locationModalDismissed, setLocationModalDismissed] = useState(false);
   104→  const [userPhotoURL, setUserPhotoURL] = useState<string>("");
   105→
   106→  // Handle URL params for PWA shortcuts
   107→  useEffect(() => {
   108→    if (typeof window === "undefined") return;
   109→    const params = new URLSearchParams(window.location.search);
   110→    const tab = params.get("tab");
   111→    if (tab && (NAV_TABS.some(t => t.id === tab) || tab === "profile" || tab === "deposit" || tab === "purchased" || tab === "submit-network")) {
   112→      setActiveTab(tab);
   113→    }
   114→  }, []);
   115→
   116→  useEffect(() => {
   117→    const unsub = onAuthStateChanged(auth, async (u) => {
   118→      setUser(u);
   119→      if (u) {
   120→        try {
   121→          const roleSnap = await get(ref(db, `users/${u.uid}/role`));
   122→          const role = roleSnap.val() || "user";
   123→          setUserRole(role);
   124→          setIsAdmin(role === "admin");
   125→          const nameSnap = await get(ref(db, `users/${u.uid}/displayName`));
   126→          setUserName(nameSnap.val() || u.displayName || u.email?.split("@")[0] || "م");
   127→          // Check if user has provinceId, show location modal if not
   128→          const provinceSnap = await get(ref(db, `users/${u.uid}/provinceId`));
   129→          const hasProvince = provinceSnap.exists() && provinceSnap.val();
   130→          if (!hasProvince && !locationModalDismissed) {
   131→            setShowLocationModal(true);
   132→          }
   133→          if (role === "network_manager") {
   134→            const netSnap = await get(ref(db, `users/${u.uid}/managedNetwork`));
   135→            setManagedNetwork(netSnap.val() || "");
   136→          }
   137→        } catch {
   138→          setIsAdmin(false);
   139→          setUserRole("user");
   140→        }
   141→      } else {
   142→        setIsAdmin(false);
   143→        setUserRole("user");
   144→        setManagedNetwork("");
   145→        setUserName("");
   146→      }
   147→      setLoading(false);
   148→    });
   149→    return () => unsub();
   150→  }, []);
   151→
   152→  useEffect(() => {
   153→    if (!user) return;
   154→    const roleUnsub = onValue(ref(db, `users/${user.uid}/role`), (snap) => {
   155→      const role = snap.val() || "user";
   156→      setUserRole(role);
   157→      setIsAdmin(role === "admin");
   158→      if (role === "network_manager") {
   159→        get(ref(db, `users/${user.uid}/managedNetwork`)).then((netSnap) => {
   160→          setManagedNetwork(netSnap.val() || "");
   161→        });
   162→      } else {
   163→        setManagedNetwork("");
   164→      }
   165→    });
   166→    const nameUnsub = onValue(ref(db, `users/${user.uid}/displayName`), (snap) => {
   167→      if (snap.val()) setUserName(snap.val());
   168→    });
   169→    return () => { roleUnsub(); nameUnsub(); };
   170→  }, [user]);
   171→
   172→  // Listen for user photo URL
   173→  useEffect(() => {
   174→    if (!user) return;
   175→    const photoUnsub = onValue(ref(db, `users/${user.uid}/photoURL`), (snap) => {
   176→      if (snap.val()) setUserPhotoURL(snap.val());
   177→      else setUserPhotoURL("");
   178→    });
   179→    return () => photoUnsub();
   180→  }, [user]);
   181→
   182→  // Enhanced splash screen with progress bar animation
   183→  useEffect(() => {
   184→    // Animate progress bar
   185→    const progressInterval = setInterval(() => {
   186→      setSplashProgress((prev) => {
   187→        if (prev >= 100) {
   188→          clearInterval(progressInterval);
   189→          return 100;
   190→        }
   191→        // Non-linear progress: fast at start, slows down, then completes
   192→        const increment = prev < 30 ? 8 : prev < 60 ? 5 : prev < 85 ? 3 : 1;
   193→        return Math.min(prev + increment, 100);
   194→      });
   195→    }, 100);
   196→
   197→    // Initialize notifications during splash
   198→    initNotifications();
   199→
   200→    // Start splash exit after delay
   201→    const exitTimer = setTimeout(() => {
   202→      setSplashExiting(true);
   203→    }, 2000);
   204→
   205→    const doneTimer = setTimeout(() => {
   206→      setSplashDone(true);
   207→      // Check if onboarding should show
   208→      if (typeof window !== "undefined" && !localStorage.getItem("applenet_onboarding_done")) {
   209→        setShowOnboarding(true);
   210→      }
   211→    }, 2500);
   212→
   213→    return () => {
   214→      clearInterval(progressInterval);
   215→      clearTimeout(exitTimer);
   216→      clearTimeout(doneTimer);
   217→    };
   218→  }, []);
   219→
   220→  const handleOnboardingComplete = useCallback(() => {
   221→    setShowOnboarding(false);
   222→  }, []);
   223→
   224→  // Enhanced Splash Screen with progress bar, version, and shimmer
   225→  if (!splashDone) {
   226→    return (
   227→      <motion.div
   228→        initial={{ opacity: 1 }}
   229→        animate={{ opacity: splashExiting ? 0 : 1 }}
   230→        transition={{ duration: 0.5, ease: "easeInOut" }}
   231→        className="fixed inset-0 bg-white dark:bg-slate-900 flex flex-col items-center justify-center"
   232→        dir={isRTL ? "rtl" : "ltr"}
   233→      >
   234→        <motion.div
   235→          initial={{ scale: 0.5, opacity: 0 }}
   236→          animate={{ scale: 1, opacity: 1 }}
   237→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
   238→          className="flex flex-col items-center"
   239→        >
   240→          {/* App Icon with shimmer */}
   241→          <div className="relative w-24 h-24 rounded-3xl overflow-hidden shadow-xl mb-6 border-2 border-[#E8F5E9] splash-shimmer">
   242→            <img src="/images/IMG_20260527_220851.jpg" alt="Apple.NET" className="w-full h-full object-cover" />
   243→          </div>
   244→          <AppleNetLogo size="lg" />
   245→
   246→          {/* Loading indicator */}
   247→          <motion.div
   248→            initial={{ opacity: 0 }}
   249→            animate={{ opacity: 1 }}
   250→            transition={{ delay: 0.5 }}
   251→            className="mt-6 flex items-center gap-2"
   252→          >
   253→            <div className="w-5 h-5 border-2 border-[#1B7A3D]/30 border-t-[#1B7A3D] rounded-full animate-spin" />
   254→            <span className="text-sm text-gray-400 dark:text-slate-500 font-bold">{t("splash.loading")}</span>
   255→          </motion.div>
   256→        </motion.div>
   257→
   258→        {/* Progress bar */}
   259→        <div className="absolute bottom-16 left-8 right-8 max-w-xs mx-auto">
   260→          <div className="h-1.5 bg-gray-100 rounded-full overflow-hidden">
   261→            <motion.div
   262→              initial={{ width: "0%" }}
   263→              animate={{ width: `${splashProgress}%` }}
   264→              transition={{ duration: 0.3, ease: "easeOut" }}
   265→              className="h-full bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] rounded-full"
   266→            />
   267→          </div>
   268→        </div>
   269→
   270→        {/* Version display */}
   271→        <div className="absolute bottom-8 text-center">
   272→          <span className="text-[10px] text-gray-300 font-mono">v{APP_VERSION}</span>
   273→        </div>
   274→      </motion.div>
   275→    );
   276→  }
   277→
   278→  // Onboarding Flow
   279→  if (showOnboarding) {
   280→    return (
   281→      <OnboardingFlow
   282→        onComplete={handleOnboardingComplete}
   283→        onLoginClick={() => { setShowOnboarding(false); setAuthMode("login"); setShowAuth(true); }}
   284→        onRegisterClick={() => { setShowOnboarding(false); setAuthMode("register"); setShowAuth(true); }}
   285→      />
   286→    );
   287→  }
   288→
   289→  return (
   290→    <div id="app-shell" className="font-sans bg-gray-50 dark:bg-slate-900" dir={isRTL ? "rtl" : "ltr"}>
   291→
   292→      {/* ===== App Update Banner ===== */}
   293→      <AppUpdateBanner />
   294→      <IOSInstallBanner />
   295→
   296→      {/* ===== STATUS BAR SIMULATION ===== */}
   297→      <div className="bg-white h-1 flex-shrink-0" />
   298→
   299→      {/* ===== LOCATION MODAL ===== */}
   300→      <AnimatePresence>
   301→        {showLocationModal && user && !locationModalDismissed && (
   302→          <motion.div
   303→            key="location-modal-overlay"
   304→            initial={{ opacity: 0 }}
   305→            animate={{ opacity: 1 }}
   306→            exit={{ opacity: 0 }}
   307→            className="fixed inset-0 z-[200] flex items-center justify-center bg-black/50 backdrop-blur-sm p-4"
   308→            dir="rtl"
   309→          >
   310→            <motion.div
   311→              key="location-modal-card"
   312→              initial={{ scale: 0.85, opacity: 0, y: 30 }}
   313→              animate={{ scale: 1, opacity: 1, y: 0 }}
   314→              exit={{ scale: 0.85, opacity: 0, y: 30 }}
   315→              transition={{ type: "spring", stiffness: 300, damping: 25 }}
   316→              className="w-full max-w-sm bg-white rounded-3xl overflow-hidden shadow-2xl"
   317→            >
   318→              {/* Green Header */}
   319→              <div className="bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] p-5 text-center relative overflow-hidden">
   320→                <div className="absolute top-0 left-0 w-24 h-24 bg-white/5 rounded-full -translate-x-1/2 -translate-y-1/2" />
   321→                <div className="absolute bottom-0 right-0 w-16 h-16 bg-white/5 rounded-full translate-x-1/3 translate-y-1/3" />
   322→                <div className="relative z-10">
   323→                  <div className="w-14 h-14 rounded-2xl bg-white/20 flex items-center justify-center mx-auto mb-3">
   324→                    <MapPin className="w-7 h-7 text-white" />
   325→                  </div>
   326→                  <h2 className="text-xl font-black text-white mb-1">{t("location.title")}</h2>
   327→                  <p className="text-white/70 text-xs leading-relaxed">{t("location.subtitle")}</p>
   328→                </div>
   329→              </div>
   330→
   331→              {/* Form */}
   332→              <div className="p-5 space-y-4">
   333→                {/* Province */}
   334→                <div>
   335→                  <label className="block text-xs text-gray-500 mb-1.5 font-bold">{t("location.province")}</label>
   336→                  <div className="relative">
   337→                    <Globe className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   338→                    <select
   339→                      value={locationModalProvinceId}
   340→                      onChange={(e) => { setLocationModalProvinceId(e.target.value); setLocationModalDistrict(""); }}
   341→                      className="w-full bg-gray-50 border border-gray-200 text-gray-900 rounded-xl h-12 text-sm pr-10 pl-4 focus:border-[#1B7A3D] focus:ring-[#1B7A3D] appearance-none"
   342→                    >
   343→                      <option value="">{t("location.selectProvince")}</option>
   344→                      {PROVINCES.map(p => (
   345→                        <option key={p.id} value={p.id}>{lang === "en" && p.nameEn ? p.nameEn : p.name}</option>
   346→                      ))}
   347→                    </select>
   348→                  </div>
   349→                </div>
   350→
   351→                {/* District */}
   352→                <div>
   353→                  <label className="block text-xs text-gray-500 mb-1.5 font-bold">{t("location.district")}</label>
   354→                  <div className="relative">
   355→                    <Building2 className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   356→                    <select
   357→                      value={locationModalDistrict}
   358→                      onChange={(e) => setLocationModalDistrict(e.target.value)}
   359→                      disabled={!locationModalProvinceId}
   360→                      className="w-full bg-gray-50 border border-gray-200 text-gray-900 rounded-xl h-12 text-sm pr-10 pl-4 focus:border-[#1B7A3D] focus:ring-[#1B7A3D] appearance-none disabled:opacity-50 disabled:cursor-not-allowed"
   361→                    >
   362→                      <option value="">{locationModalProvinceId ? t("location.selectDistrict") : t("location.selectProvinceFirst")}</option>
   363→                      {locationModalProvinceId && (() => {
   364→                        const districtNames = lang === "en" ? getDistrictsEn(locationModalProvinceId) : getDistricts(locationModalProvinceId);
   365→                        const districtValues = getDistricts(locationModalProvinceId);
   366→                        return districtValues.map((d, i) => (
   367→                          <option key={d} value={d}>{districtNames[i]}</option>
   368→                        ));
   369→                      })()}
   370→                    </select>
   371→                  </div>
   372→                </div>
   373→
   374→                {/* Buttons */}
   375→                <div className="space-y-2 pt-1">
   376→                  <Button
   377→                    onClick={async () => {
   378→                      if (!locationModalProvinceId) {
   379→                        toast.error(t("location.selectProvinceFirst"));
   380→                        return;
   381→                      }
   382→                      setLocationModalSaving(true);
   383→                      try {
   384→                        const provinceObj = PROVINCES.find(p => p.id === locationModalProvinceId);
   385→                        await update(ref(db, `users/${user.uid}`), {
   386→                          provinceId: locationModalProvinceId,
   387→                          provinceName: provinceObj?.name || null,
   388→                          district: locationModalDistrict || null,
   389→                        });
   390→                        toast.success(t("location.savedSuccess"));
   391→                        setShowLocationModal(false);
   392→                        setLocationModalDismissed(true);
   393→                      } catch {
   394→                        toast.error(t("location.saveError"));
   395→                      }
   396→                      setLocationModalSaving(false);
   397→                    }}
   398→                    disabled={locationModalSaving}
   399→                    className="w-full bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] hover:from-[#165E30] hover:to-[#134D28] text-white font-bold rounded-2xl h-12 btn-green-shadow disabled:opacity-70"
   400→                  >
   401→                    {locationModalSaving ? (
   402→                      <span className="flex items-center gap-2">
   403→                        <div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />
   404→                        {t("location.saving")}
   405→                      </span>
   406→                    ) : t("location.save")}
   407→                  </Button>
   408→                  <Button
   409→                    variant="ghost"
   410→                    onClick={() => {
   411→                      setShowLocationModal(false);
   412→                      setLocationModalDismissed(true);
   413→                    }}
   414→                    className="w-full text-gray-400 font-bold rounded-xl h-10 text-sm hover:text-gray-600"
   415→                  >
   416→                    {t("location.skip")}
   417→                  </Button>
   418→                </div>
   419→              </div>
   420→            </motion.div>
   421→          </motion.div>
   422→        )}
   423→      </AnimatePresence>
   424→
   425→      {/* ===== APP HEADER ===== */}
   426→      <header className="flex-shrink-0 app-header border-b border-gray-100/50 z-40">
   427→        <div className="max-w-lg mx-auto px-4 py-2.5 flex items-center justify-between">
   428→          {/* Menu Button */}
   429→          <motion.button
   430→            whileTap={{ scale: 0.85 }}
   431→            onClick={() => setShowMenu(!showMenu)}
   432→            className="w-10 h-10 rounded-2xl bg-[#E8F5E9] flex items-center justify-center hover:bg-green-100 transition-colors haptic-press"
   433→          >
   434→            <Menu className="w-5 h-5 text-[#1B7A3D]" />
   435→          </motion.button>
   436→
   437→          {/* App Logo */}
   438→          <AppleNetLogo size="sm" />
   439→
   440→          {/* Right Actions */}
   441→          <div className="flex items-center gap-1">
   442→            <ThemeToggle />
   443→            <LanguageToggle />
   444→            {user && <NotificationCenter uid={user.uid} isAdmin={isAdmin} />}
   445→            {user ? (
   446→              <motion.button
   447→                whileTap={{ scale: 0.85 }}
   448→                onClick={async () => { await signOut(auth); toast.success(t("auth.logout")); }}
   449→                className="w-10 h-10 rounded-2xl bg-red-50 dark:bg-red-900/20 flex items-center justify-center hover:bg-red-100 dark:hover:bg-red-900/30 transition-colors haptic-press"
   450→              >
   451→                <LogOut className="w-4 h-4 text-red-500" />
   452→              </motion.button>
   453→            ) : (
   454→              <motion.button
   455→                whileTap={{ scale: 0.85 }}
   456→                onClick={() => setShowAuth(true)}
   457→                className="w-10 h-10 rounded-2xl bg-[#E8F5E9] dark:bg-green-900/30 flex items-center justify-center hover:bg-green-100 dark:hover:bg-green-900/40 transition-colors haptic-press"
   458→              >
   459→                <LogIn className="w-4 h-4 text-[#1B7A3D]" />
   460→              </motion.button>
   461→            )}
   462→          </div>
   463→        </div>
   464→      </header>
   465→
   466→      {/* ===== SIDE MENU (Drawer) ===== */}
   467→      <AnimatePresence>
   468→        {showMenu && (
   469→          <>
   470→            <motion.div
   471→              initial={{ opacity: 0 }}
   472→              animate={{ opacity: 1 }}
   473→              exit={{ opacity: 0 }}
   474→              className="fixed inset-0 bg-black/40 z-[100]"
   475→              onClick={() => setShowMenu(false)}
   476→            />
   477→            <motion.div
   478→              initial={{ x: 288, opacity: 0 }}
   479→              animate={{ x: 0, opacity: 1 }}
   480→              exit={{ x: 288, opacity: 0 }}
   481→              transition={iOSSpring.gentle}
   482→              className="fixed top-0 right-0 bottom-0 w-[288px] bg-white dark:bg-slate-900 z-[101] shadow-2xl border-l border-gray-100 dark:border-slate-700/50"
   483→            >
   484→              <div className="p-5">
   485→                {/* Menu Header */}
   486→                <div className="flex items-center justify-between mb-5">
   487→                  <div className="w-12 h-12 rounded-2xl overflow-hidden border-2 border-[#E8F5E9] dark:border-green-900/30">
   488→                    <img src="/images/IMG_20260527_220851.jpg" alt="Apple.NET" className="w-full h-full object-cover" />
   489→                  </div>
   490→                  <button onClick={() => setShowMenu(false)} className="w-8 h-8 rounded-full bg-gray-100 dark:bg-slate-700 flex items-center justify-center">
   491→                    <X className="w-4 h-4 text-gray-400 dark:text-slate-400" />
   492→                  </button>
   493→                </div>
   494→
   495→                {/* User Card */}
