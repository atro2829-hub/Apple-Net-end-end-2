    71→
    72→export default function AppleNetApp() {
    73→  const { t, isRTL, lang } = useLanguage();
    74→
    75→  const NAV_TABS = [
    76→    { id: "home", icon: Home, label: t("nav.home") },
    77→    { id: "cards", icon: Wifi, label: t("nav.cards") },
    78→    { id: "starlink", icon: Satellite, label: "Starlink" },
    79→    { id: "chat", icon: MessageCircle, label: t("nav.chat") },
    80→    { id: "credit", icon: Wallet, label: t("nav.credit") },
    81→  ];
    82→
    83→  const [user, setUser] = useState<FirebaseUser | null>(null);
    84→  const [isAdmin, setIsAdmin] = useState(false);
    85→  const [userRole, setUserRole] = useState<string>("user");
    86→  const [managedNetwork, setManagedNetwork] = useState<string>("");
    87→  const [activeTab, setActiveTab] = useState("home");
    88→  const [showAuth, setShowAuth] = useState(false);
    89→  const [authMode, setAuthMode] = useState<"login" | "register">("login");
    90→  const [showAdmin, setShowAdmin] = useState(false);
    91→  const [loading, setLoading] = useState(true);
    92→  const [showMenu, setShowMenu] = useState(false);
    93→  const [splashDone, setSplashDone] = useState(false);
    94→  const [splashExiting, setSplashExiting] = useState(false);
    95→  const [userName, setUserName] = useState<string>("");
    96→  const [showOnboarding, setShowOnboarding] = useState(false);
    97→  const [splashProgress, setSplashProgress] = useState(0);
    98→  const [showLocationModal, setShowLocationModal] = useState(false);
    99→  const [locationModalProvinceId, setLocationModalProvinceId] = useState("");
   100→  const [locationModalDistrict, setLocationModalDistrict] = useState("");
   101→  const [locationModalSaving, setLocationModalSaving] = useState(false);
   102→  const [locationModalDismissed, setLocationModalDismissed] = useState(false);
   103→  const [userPhotoURL, setUserPhotoURL] = useState<string>("");
   104→
   105→  // Handle URL params for PWA shortcuts
   106→  useEffect(() => {
   107→    if (typeof window === "undefined") return;
   108→    const params = new URLSearchParams(window.location.search);
   109→    const tab = params.get("tab");
   110→    if (tab && (NAV_TABS.some(t => t.id === tab) || tab === "profile" || tab === "deposit" || tab === "purchased" || tab === "submit-network")) {
   111→      setActiveTab(tab);
   112→    }
   113→  }, []);
   114→
   115→  useEffect(() => {
   116→    const unsub = onAuthStateChanged(auth, async (u) => {
   117→      setUser(u);
   118→      if (u) {
   119→        try {
   120→          const roleSnap = await get(ref(db, `users/${u.uid}/role`));
   121→          const role = roleSnap.val() || "user";
   122→          setUserRole(role);
   123→          setIsAdmin(role === "admin");
   124→          const nameSnap = await get(ref(db, `users/${u.uid}/displayName`));
   125→          setUserName(nameSnap.val() || u.displayName || u.email?.split("@")[0] || "م");
   126→          // Check if user has provinceId, show location modal if not
   127→          const provinceSnap = await get(ref(db, `users/${u.uid}/provinceId`));
   128→          const hasProvince = provinceSnap.exists() && provinceSnap.val();
   129→          if (!hasProvince && !locationModalDismissed) {
   130→            setShowLocationModal(true);
   131→          }
   132→          if (role === "network_manager") {
   133→            const netSnap = await get(ref(db, `users/${u.uid}/managedNetwork`));
   134→            setManagedNetwork(netSnap.val() || "");
   135→          }
   136→        } catch {
   137→          setIsAdmin(false);
   138→          setUserRole("user");
   139→        }
   140→      } else {
   141→        setIsAdmin(false);
   142→        setUserRole("user");
   143→        setManagedNetwork("");
   144→        setUserName("");
   145→      }
   146→      setLoading(false);
   147→    });
   148→    return () => unsub();
   149→  }, []);
   150→
   151→  useEffect(() => {
   152→    if (!user) return;
   153→    const roleUnsub = onValue(ref(db, `users/${user.uid}/role`), (snap) => {
   154→      const role = snap.val() || "user";
   155→      setUserRole(role);
   156→      setIsAdmin(role === "admin");
   157→      if (role === "network_manager") {
   158→        get(ref(db, `users/${user.uid}/managedNetwork`)).then((netSnap) => {
   159→          setManagedNetwork(netSnap.val() || "");
   160→        });
   161→      } else {
   162→        setManagedNetwork("");
   163→      }
   164→    });
   165→    const nameUnsub = onValue(ref(db, `users/${user.uid}/displayName`), (snap) => {
   166→      if (snap.val()) setUserName(snap.val());
   167→    });
   168→    return () => { roleUnsub(); nameUnsub(); };
   169→  }, [user]);
   170→
   171→  // Listen for user photo URL
   172→  useEffect(() => {
   173→    if (!user) return;
   174→    const photoUnsub = onValue(ref(db, `users/${user.uid}/photoURL`), (snap) => {
   175→      if (snap.val()) setUserPhotoURL(snap.val());
   176→      else setUserPhotoURL("");
   177→    });
   178→    return () => photoUnsub();
   179→  }, [user]);
   180→
   181→  // Enhanced splash screen with progress bar animation
   182→  useEffect(() => {
   183→    // Animate progress bar
   184→    const progressInterval = setInterval(() => {
   185→      setSplashProgress((prev) => {
   186→        if (prev >= 100) {
   187→          clearInterval(progressInterval);
   188→          return 100;
   189→        }
   190→        // Non-linear progress: fast at start, slows down, then completes
   191→        const increment = prev < 30 ? 8 : prev < 60 ? 5 : prev < 85 ? 3 : 1;
   192→        return Math.min(prev + increment, 100);
   193→      });
   194→    }, 100);
   195→
   196→    // Initialize notifications during splash
   197→    initNotifications();
   198→
   199→    // Start splash exit after delay
   200→    const exitTimer = setTimeout(() => {
   201→      setSplashExiting(true);
   202→    }, 2000);
   203→
   204→    const doneTimer = setTimeout(() => {
   205→      setSplashDone(true);
   206→      // Check if onboarding should show
   207→      if (typeof window !== "undefined" && !localStorage.getItem("applenet_onboarding_done")) {
   208→        setShowOnboarding(true);
   209→      }
   210→    }, 2500);
   211→
   212→    return () => {
   213→      clearInterval(progressInterval);
   214→      clearTimeout(exitTimer);
   215→      clearTimeout(doneTimer);
   216→    };
   217→  }, []);
   218→
   219→  const handleOnboardingComplete = useCallback(() => {
   220→    setShowOnboarding(false);
   221→  }, []);
   222→
   223→  // Enhanced Splash Screen with progress bar, version, and shimmer
   224→  if (!splashDone) {
   225→    return (
   226→      <motion.div
   227→        initial={{ opacity: 1 }}
   228→        animate={{ opacity: splashExiting ? 0 : 1 }}
   229→        transition={{ duration: 0.5, ease: "easeInOut" }}
   230→        className="fixed inset-0 bg-white dark:bg-slate-900 flex flex-col items-center justify-center"
   231→        dir={isRTL ? "rtl" : "ltr"}
   232→      >
   233→        <motion.div
   234→          initial={{ scale: 0.5, opacity: 0 }}
   235→          animate={{ scale: 1, opacity: 1 }}
   236→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
   237→          className="flex flex-col items-center"
   238→        >
   239→          {/* App Icon with shimmer */}
   240→          <div className="relative w-24 h-24 rounded-3xl overflow-hidden shadow-xl mb-6 border-2 border-[#E8F5E9] splash-shimmer">
   241→            <img src="/images/IMG_20260527_220851.jpg" alt="Apple.NET" className="w-full h-full object-cover" />
   242→          </div>
   243→          <AppleNetLogo size="lg" />
   244→
   245→          {/* Loading indicator */}
   246→          <motion.div
   247→            initial={{ opacity: 0 }}
   248→            animate={{ opacity: 1 }}
   249→            transition={{ delay: 0.5 }}
   250→            className="mt-6 flex items-center gap-2"
   251→          >
   252→            <div className="w-5 h-5 border-2 border-[#1B7A3D]/30 border-t-[#1B7A3D] rounded-full animate-spin" />
   253→            <span className="text-sm text-gray-400 dark:text-slate-500 font-bold">{t("splash.loading")}</span>
   254→          </motion.div>
   255→        </motion.div>
   256→
   257→        {/* Progress bar */}
   258→        <div className="absolute bottom-16 left-8 right-8 max-w-xs mx-auto">
   259→          <div className="h-1.5 bg-gray-100 rounded-full overflow-hidden">
   260→            <motion.div
   261→              initial={{ width: "0%" }}
   262→              animate={{ width: `${splashProgress}%` }}
   263→              transition={{ duration: 0.3, ease: "easeOut" }}
   264→              className="h-full bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] rounded-full"
   265→            />
   266→          </div>
   267→        </div>
   268→
   269→        {/* Version display */}
   270→        <div className="absolute bottom-8 text-center">
   271→          <span className="text-[10px] text-gray-300 font-mono">v{APP_VERSION}</span>
   272→        </div>
   273→      </motion.div>
   274→    );
   275→  }
   276→
   277→  // Onboarding Flow
   278→  if (showOnboarding) {
   279→    return (
   280→      <OnboardingFlow
   281→        onComplete={handleOnboardingComplete}
   282→        onLoginClick={() => { setShowOnboarding(false); setAuthMode("login"); setShowAuth(true); }}
   283→        onRegisterClick={() => { setShowOnboarding(false); setAuthMode("register"); setShowAuth(true); }}
   284→      />
   285→    );
   286→  }
   287→
   288→  return (
   289→    <div id="app-shell" className="font-sans bg-gray-50 dark:bg-slate-900" dir={isRTL ? "rtl" : "ltr"}>
   290→
   291→      {/* ===== App Update Banner ===== */}
   292→      <AppUpdateBanner />
   293→      <IOSInstallBanner />
   294→
   295→      {/* ===== STATUS BAR SIMULATION ===== */}
   296→      <div className="bg-white h-1 flex-shrink-0" />
   297→
   298→      {/* ===== LOCATION MODAL ===== */}
   299→      <AnimatePresence>
   300→        {showLocationModal && user && !locationModalDismissed && (
   301→          <motion.div
   302→            key="location-modal-overlay"
   303→            initial={{ opacity: 0 }}
   304→            animate={{ opacity: 1 }}
   305→            exit={{ opacity: 0 }}
   306→            className="fixed inset-0 z-[200] flex items-center justify-center bg-black/50 backdrop-blur-sm p-4"
   307→            dir="rtl"
   308→          >
   309→            <motion.div
   310→              key="location-modal-card"
   311→              initial={{ scale: 0.85, opacity: 0, y: 30 }}
   312→              animate={{ scale: 1, opacity: 1, y: 0 }}
   313→              exit={{ scale: 0.85, opacity: 0, y: 30 }}
   314→              transition={{ type: "spring", stiffness: 300, damping: 25 }}
   315→              className="w-full max-w-sm bg-white rounded-3xl overflow-hidden shadow-2xl"
   316→            >
   317→              {/* Green Header */}
   318→              <div className="bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] p-5 text-center relative overflow-hidden">
   319→                <div className="absolute top-0 left-0 w-24 h-24 bg-white/5 rounded-full -translate-x-1/2 -translate-y-1/2" />
   320→                <div className="absolute bottom-0 right-0 w-16 h-16 bg-white/5 rounded-full translate-x-1/3 translate-y-1/3" />
   321→                <div className="relative z-10">
   322→                  <div className="w-14 h-14 rounded-2xl bg-white/20 flex items-center justify-center mx-auto mb-3">
   323→                    <MapPin className="w-7 h-7 text-white" />
   324→                  </div>
   325→                  <h2 className="text-xl font-black text-white mb-1">{t("location.title")}</h2>
   326→                  <p className="text-white/70 text-xs leading-relaxed">{t("location.subtitle")}</p>
   327→                </div>
   328→              </div>
   329→
   330→              {/* Form */}
   331→              <div className="p-5 space-y-4">
   332→                {/* Province */}
   333→                <div>
   334→                  <label className="block text-xs text-gray-500 mb-1.5 font-bold">{t("location.province")}</label>
   335→                  <div className="relative">
   336→                    <Globe className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   337→                    <select
   338→                      value={locationModalProvinceId}
   339→                      onChange={(e) => { setLocationModalProvinceId(e.target.value); setLocationModalDistrict(""); }}
   340→                      className="w-full bg-gray-50 border border-gray-200 text-gray-900 rounded-xl h-12 text-sm pr-10 pl-4 focus:border-[#1B7A3D] focus:ring-[#1B7A3D] appearance-none"
   341→                    >
   342→                      <option value="">{t("location.selectProvince")}</option>
   343→                      {PROVINCES.map(p => (
   344→                        <option key={p.id} value={p.id}>{lang === "en" && p.nameEn ? p.nameEn : p.name}</option>
   345→                      ))}
   346→                    </select>
   347→                  </div>
   348→                </div>
   349→
   350→                {/* District */}
   351→                <div>
   352→                  <label className="block text-xs text-gray-500 mb-1.5 font-bold">{t("location.district")}</label>
   353→                  <div className="relative">
   354→                    <Building2 className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   355→                    <select
   356→                      value={locationModalDistrict}
   357→                      onChange={(e) => setLocationModalDistrict(e.target.value)}
   358→                      disabled={!locationModalProvinceId}
   359→                      className="w-full bg-gray-50 border border-gray-200 text-gray-900 rounded-xl h-12 text-sm pr-10 pl-4 focus:border-[#1B7A3D] focus:ring-[#1B7A3D] appearance-none disabled:opacity-50 disabled:cursor-not-allowed"
   360→                    >
   361→                      <option value="">{locationModalProvinceId ? t("location.selectDistrict") : t("location.selectProvinceFirst")}</option>
   362→                      {locationModalProvinceId && (() => {
   363→                        const districtNames = lang === "en" ? getDistrictsEn(locationModalProvinceId) : getDistricts(locationModalProvinceId);
   364→                        const districtValues = getDistricts(locationModalProvinceId);
   365→                        return districtValues.map((d, i) => (
   366→                          <option key={d} value={d}>{districtNames[i]}</option>
   367→                        ));
   368→                      })()}
   369→                    </select>
   370→                  </div>
   371→                </div>
   372→
   373→                {/* Buttons */}
   374→                <div className="space-y-2 pt-1">
   375→                  <Button
   376→                    onClick={async () => {
   377→                      if (!locationModalProvinceId) {
   378→                        toast.error(t("location.selectProvinceFirst"));
   379→                        return;
   380→                      }
   381→                      setLocationModalSaving(true);
   382→                      try {
   383→                        const provinceObj = PROVINCES.find(p => p.id === locationModalProvinceId);
   384→                        await update(ref(db, `users/${user.uid}`), {
   385→                          provinceId: locationModalProvinceId,
   386→                          provinceName: provinceObj?.name || null,
   387→                          district: locationModalDistrict || null,
   388→                        });
   389→                        toast.success(t("location.savedSuccess"));
   390→                        setShowLocationModal(false);
   391→                        setLocationModalDismissed(true);
   392→                      } catch {
   393→                        toast.error(t("location.saveError"));
   394→                      }
   395→                      setLocationModalSaving(false);
   396→                    }}
   397→                    disabled={locationModalSaving}
   398→                    className="w-full bg-gradient-to-l from-[#1B7A3D] to-[#22A24D] hover:from-[#165E30] hover:to-[#134D28] text-white font-bold rounded-2xl h-12 btn-green-shadow disabled:opacity-70"
   399→                  >
   400→                    {locationModalSaving ? (
   401→                      <span className="flex items-center gap-2">
   402→                        <div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />
   403→                        {t("location.saving")}
   404→                      </span>
   405→                    ) : t("location.save")}
   406→                  </Button>
   407→                  <Button
   408→                    variant="ghost"
   409→                    onClick={() => {
   410→                      setShowLocationModal(false);
   411→                      setLocationModalDismissed(true);
   412→                    }}
   413→                    className="w-full text-gray-400 font-bold rounded-xl h-10 text-sm hover:text-gray-600"
   414→                  >
   415→                    {t("location.skip")}
   416→                  </Button>
   417→                </div>
   418→              </div>
   419→            </motion.div>
   420→          </motion.div>
   421→        )}
   422→      </AnimatePresence>
   423→
   424→      {/* ===== APP HEADER ===== */}
   425→      <header className="flex-shrink-0 app-header border-b border-gray-100/50 z-40">
   426→        <div className="max-w-lg mx-auto px-4 py-2.5 flex items-center justify-between">
   427→          {/* Menu Button */}
   428→          <motion.button
   429→            whileTap={{ scale: 0.85 }}
   430→            onClick={() => setShowMenu(!showMenu)}
   431→            className="w-10 h-10 rounded-2xl bg-[#E8F5E9] flex items-center justify-center hover:bg-green-100 transition-colors haptic-press"
   432→          >
   433→            <Menu className="w-5 h-5 text-[#1B7A3D]" />
   434→          </motion.button>
   435→
   436→          {/* App Logo */}
   437→          <AppleNetLogo size="sm" />
   438→
   439→          {/* Right Actions */}
   440→          <div className="flex items-center gap-1">
   441→            <ThemeToggle />
   442→            <LanguageToggle />
   443→            {user && <NotificationCenter uid={user.uid} isAdmin={isAdmin} />}
   444→            {user ? (
   445→              <motion.button
   446→                whileTap={{ scale: 0.85 }}
   447→                onClick={async () => { await signOut(auth); toast.success(t("auth.logout")); }}
   448→                className="w-10 h-10 rounded-2xl bg-red-50 dark:bg-red-900/20 flex items-center justify-center hover:bg-red-100 dark:hover:bg-red-900/30 transition-colors haptic-press"
   449→              >
   450→                <LogOut className="w-4 h-4 text-red-500" />
   451→              </motion.button>
   452→            ) : (
   453→              <motion.button
   454→                whileTap={{ scale: 0.85 }}
   455→                onClick={() => setShowAuth(true)}
   456→                className="w-10 h-10 rounded-2xl bg-[#E8F5E9] dark:bg-green-900/30 flex items-center justify-center hover:bg-green-100 dark:hover:bg-green-900/40 transition-colors haptic-press"
   457→              >
   458→                <LogIn className="w-4 h-4 text-[#1B7A3D]" />
   459→              </motion.button>
   460→            )}
   461→          </div>
   462→        </div>
   463→      </header>
   464→
   465→      {/* ===== SIDE MENU (Drawer) ===== */}
   466→      <AnimatePresence>
   467→        {showMenu && (
   468→          <>
   469→            <motion.div
   470→              initial={{ opacity: 0 }}
   471→              animate={{ opacity: 1 }}
   472→              exit={{ opacity: 0 }}
   473→              className="fixed inset-0 bg-black/40 z-[100]"
   474→              onClick={() => setShowMenu(false)}
   475→            />
   476→            <motion.div
   477→              initial={{ x: 288, opacity: 0 }}
   478→              animate={{ x: 0, opacity: 1 }}
   479→              exit={{ x: 288, opacity: 0 }}
   480→              transition={iOSSpring.gentle}
   481→              className="fixed top-0 right-0 bottom-0 w-[288px] bg-white dark:bg-slate-900 z-[101] shadow-2xl border-l border-gray-100 dark:border-slate-700/50"
   482→            >
   483→              <div className="p-5">
   484→                {/* Menu Header */}
   485→                <div className="flex items-center justify-between mb-5">
   486→                  <div className="w-12 h-12 rounded-2xl overflow-hidden border-2 border-[#E8F5E9] dark:border-green-900/30">
   487→                    <img src="/images/IMG_20260527_220851.jpg" alt="Apple.NET" className="w-full h-full object-cover" />
   488→                  </div>
   489→                  <button onClick={() => setShowMenu(false)} className="w-8 h-8 rounded-full bg-gray-100 dark:bg-slate-700 flex items-center justify-center">
   490→                    <X className="w-4 h-4 text-gray-400 dark:text-slate-400" />
   491→                  </button>
   492→                </div>
   493→
   494→                {/* User Card */}
   495→                {user && (
