   170→  const handleRedeemCode = async () => {
   171→    if (!user || !redeemCode.trim()) {
   172→      toast.error(t("credit.enterCodeFirst"));
   173→      return;
   174→    }
   175→    setIsRedeeming(true);
   176→    try {
   177→      const enteredCode = normalizeCode(redeemCode);
   178→      console.log("[Redeem] Normalized code:", enteredCode, "length:", enteredCode.length);
   179→
   180→      if (enteredCode.length < 4) {
   181→        toast.error(t("credit.codeTooShort"));
   182→        setIsRedeeming(false);
   183→        return;
   184→      }
   185→
   186→      // ─── Ensure credit record exists ──────────────────
   187→      const creditSnap = await get(ref(db, `credit/${user.uid}`));
   188→      if (!creditSnap.exists()) {
   189→        await set(ref(db, `credit/${user.uid}`), { amount: 0, updatedAt: Date.now() });
   190→        console.log("[Redeem] Created credit record for user");
   191→      }
   192→
   193→      // 1) Check shared redeem code lookup first (mass charge codes)
   194→      const sharedLookupSnap = await get(ref(db, `sharedRedeemCodeLookup/${enteredCode}`));
   195→      const sharedLookupData = sharedLookupSnap.val() as { pushId: string; amount: number; maxRedemptions: number; currentRedemptions: number; isActive: boolean } | null;
   196→
   197→      if (sharedLookupData) {
   198→        console.log("[Redeem] Found shared code:", sharedLookupData);
   199→
   200→        if (sharedLookupData.isActive === false) {
   201→          toast.error(t("credit.codeDisabled"));
   202→          setIsRedeeming(false);
   203→          return;
   204→        }
   205→
   206→        if (sharedLookupData.currentRedemptions >= sharedLookupData.maxRedemptions) {
   207→          toast.error(t("credit.codeMaxReached"));
   208→          setIsRedeeming(false);
   209→          return;
   210→        }
   211→
   212→        const redeemedBySnap = await get(ref(db, `sharedRedeemCodes/${sharedLookupData.pushId}/redeemedBy/${user.uid}`));
   213→        if (redeemedBySnap.exists()) {
   214→          toast.error(t("credit.codeAlreadyUsed"));
   215→          setIsRedeeming(false);
   216→          return;
   217→        }
   218→
   219→        const codeAmount = sharedLookupData.amount;
   220→        if (codeAmount <= 0) {
   221→          toast.error(t("credit.codeInvalid"));
   222→          setIsRedeeming(false);
   223→          return;
   224→        }
   225→
   226→        // Add credit atomically
   227→        const amountRef = ref(db, `credit/${user.uid}/amount`);
   228→        let actualAmount = codeAmount;
   229→        const creditResult = await runTransaction(amountRef, (currentAmount) => {
   230→          const bal = (currentAmount || 0) as number;
   231→          let newBal = bal + codeAmount;
   232→          if (maxBalance > 0 && newBal > maxBalance) {
   233→            actualAmount = Math.max(maxBalance - bal, 0);
   234→            if (actualAmount <= 0) return bal;
   235→            newBal = maxBalance;
   236→          }
   237→          return newBal;
   238→        });
   239→
   240→        if (!creditResult.committed || actualAmount <= 0) {
   241→          toast.error(`${t("credit.balanceCeiling")} ${maxBalance.toLocaleString()} ${t("credit.riyalShort")}`);
   242→          setIsRedeeming(false);
   243→          return;
   244→        }
   245→
   246→        // Atomically increment currentRedemptions and add user to redeemedBy
   247→        const sharedCodeRef = ref(db, `sharedRedeemCodes/${sharedLookupData.pushId}`);
   248→        const transactionResult = await runTransaction(sharedCodeRef, (currentData) => {
   249→          if (currentData === null) return null;
   250→          const data = currentData as Record<string, unknown>;
   251→          if (data.isActive === false) return;
   252→          const currentRedemptions = (data.currentRedemptions as number) || 0;
   253→          const maxRedemptions = (data.maxRedemptions as number) || 0;
   254→          if (currentRedemptions >= maxRedemptions) return;
   255→          const redeemedBy = (data.redeemedBy as Record<string, unknown>) || {};
   256→          if (redeemedBy[user.uid]) return;
   257→          return {
   258→            ...data,
   259→            currentRedemptions: currentRedemptions + 1,
   260→            redeemedBy: {
   261→              ...redeemedBy,
   262→              [user.uid]: {
   263→                uid: user.uid,
   264→                name: user.displayName || user.email || t("credit.user"),
   265→                redeemedAt: Date.now(),
   266→              },
   267→            },
   268→          };
   269→        });
   270→
   271→        if (!transactionResult.committed) {
   272→          await runTransaction(amountRef, (cur) => {
   273→            const bal = (cur || 0) as number;
   274→            return Math.max(bal - actualAmount, 0);
   275→          });
   276→          toast.error(t("credit.sharedCodeFailed"));
   277→          setIsRedeeming(false);
   278→          return;
   279→        }
   280→
   281→        // Update lookup currentRedemptions
   282→        try {
   283→          await update(ref(db, `sharedRedeemCodeLookup/${enteredCode}`), {
   284→            currentRedemptions: (sharedLookupData.currentRedemptions || 0) + 1,
   285→          });
   286→        } catch { /* non-critical */ }
   287→
   288→        await update(ref(db, `credit/${user.uid}`), { updatedAt: Date.now() });
   289→
   290→        const histRef = push(ref(db, `credit/${user.uid}/history`));
   291→        await set(histRef, {
   292→          type: "redeem",
   293→          amount: actualAmount,
   294→          description: `${t("credit.sharedCodeCharge")} (${enteredCode})`,
   295→          date: Date.now(),
   296→        });
   297→
   298→        const notifRef = push(ref(db, `notifications/${user.uid}`));
   299→        await set(notifRef, {
   300→          type: "deposit_approved",
   301→          title: t("credit.sharedCodeCharge"),
   302→          message: `${t("credit.sharedCodeChargeMsg")} ${actualAmount.toLocaleString()} ${t("credit.riyalShort")} ${t("credit.viaSharedCode")}`,
   303→          isRead: false,
   304→          createdAt: Date.now(),
   305→        });
   306→
   307→        toast.success(`${t("credit.chargedAmountSuccess")} ${actualAmount.toLocaleString()} ${t("credit.chargedSuccessFull")}`);
   308→        setRedeemSuccess({ amount: actualAmount, visible: true });
   309→        setRedeemCode("");
   310→        setShowRedeem(false);
   311→        setIsRedeeming(false);
   312→        setTimeout(() => setRedeemSuccess(null), 3000);
   313→        return;
   314→      }
   315→
   316→      // 2) Check regular redeem code lookup
   317→      let foundCodeId: string | null = null;
   318→      let codeAmount = 0;
   319→
   320→      const lookupSnap = await get(ref(db, `redeemCodeLookup/${enteredCode}`));
   321→      const lookupData = lookupSnap.val() as { pushId: string; amount: number; isUsed: boolean } | null;
   322→      console.log("[Redeem] Lookup result:", lookupData);
   323→
   324→      if (lookupData) {
   325→        if (lookupData.isUsed === true) {
   326→          toast.error(t("credit.codeUsedAlready"));
   327→          setIsRedeeming(false);
   328→          return;
   329→        }
   330→        foundCodeId = lookupData.pushId;
   331→        codeAmount = lookupData.amount || 0;
   332→      } else {
   333→        // 3) Lookup not found - search directly in redeemCodes
   334→        console.log("[Redeem] Lookup not found, searching redeemCodes directly...");
   335→        const codesSnap = await get(ref(db, "redeemCodes"));
   336→        if (codesSnap.exists()) {
   337→          const allCodes = codesSnap.val() as Record<string, { code: string; amount: number; isUsed: boolean }>;
   338→          for (const [id, val] of Object.entries(allCodes)) {
   339→            const normalizedStoredCode = normalizeCode(val.code || "");
   340→            if (normalizedStoredCode === enteredCode) {
   341→              if (val.isUsed === true) {
   342→                toast.error(t("credit.codeUsedAlready"));
   343→                setIsRedeeming(false);
   344→                return;
   345→              }
   346→              foundCodeId = id;
   347→              codeAmount = val.amount || 0;
   348→              break;
   349→            }
   350→          }
   351→        }
   352→        if (!foundCodeId) {
   353→          toast.error(t("credit.codeNotFound"));
   354→          setIsRedeeming(false);
   355→          return;
   356→        }
   357→      }
   358→
   359→      if (codeAmount <= 0) {
   360→        toast.error(t("credit.codeInvalid"));
   361→        setIsRedeeming(false);
   362→        return;
   363→      }
   364→
   365→      // 4) Charge credit using runTransaction
   366→      const amountRef = ref(db, `credit/${user.uid}/amount`);
   367→      const chargeResult = await runTransaction(amountRef, (currentAmount) => {
   368→        const bal = (currentAmount || 0) as number;
   369→        if (maxBalance > 0 && bal + codeAmount > maxBalance) {
   370→          return; // abort transaction
   371→        }
   372→        return bal + codeAmount;
   373→      });
   374→
   375→      if (!chargeResult.committed) {
   376→        toast.error(`${t("credit.balanceCeiling")} ${maxBalance.toLocaleString()} ${t("credit.riyalShort")} ${t("credit.balanceCeilingPreventsAdd")}`);
   377→        setIsRedeeming(false);
   378→        return;
   379→      }
   380→
   381→      await update(ref(db, `credit/${user.uid}`), { updatedAt: Date.now() });
   382→      const finalAmount = chargeResult.snapshot.val();
   383→      console.log("[Redeem] Credit updated successfully, new amount:", finalAmount);
   384→
   385→      // 5) Mark code as used in lookup (non-critical if fails)
   386→      try {
   387→        await update(ref(db, `redeemCodeLookup/${enteredCode}`), {
   388→          isUsed: true,
   389→          usedBy: user.uid,
   390→          usedAt: Date.now(),
   391→        });
   392→      } catch { /* lookup may not exist for older codes */ }
   393→
   394→      // 6) Mark code as used in codes table
   395→      await update(ref(db, `redeemCodes/${foundCodeId}`), {
   396→        isUsed: true,
   397→        usedBy: user.uid,
   398→        usedByName: user.displayName || user.email || t("credit.user"),
   399→        usedAt: Date.now(),
   400→      });
   401→
   402→      // 6b) Delete the code from database after successful redemption
   403→      try {
   404→        await remove(ref(db, `redeemCodes/${foundCodeId}`));
   405→        if (enteredCode) {
   406→          await remove(ref(db, `redeemCodeLookup/${enteredCode}`));
   407→        }
   408→      } catch { /* non-critical - code already processed */ }
   409→
   410→      // 7) Add to history
   411→      const histRef = push(ref(db, `credit/${user.uid}/history`));
   412→      await set(histRef, {
   413→        type: "redeem",
   414→        amount: codeAmount,
   415→        description: `${t("credit.codeCharge")} (${enteredCode})`,
   416→        date: Date.now(),
   417→      });
   418→
   419→      // 8) Send notification
   420→      const notifRef = push(ref(db, `notifications/${user.uid}`));
   421→      await set(notifRef, {
   422→        type: "deposit_approved",
   423→        title: t("credit.codeCharge"),
   424→        message: `${t("credit.codeChargeMsg")} ${codeAmount.toLocaleString()} ${t("credit.riyalShort")} ${t("credit.viaChargeCode")}`,
   425→        isRead: false,
   426→        createdAt: Date.now(),
   427→      });
   428→
   429→      toast.success(`${t("credit.chargedAmountSuccess")} ${codeAmount.toLocaleString()} ${t("credit.chargedSuccessFull")}`);
   430→      setRedeemSuccess({ amount: codeAmount, visible: true });
   431→      setRedeemCode("");
   432→      setShowRedeem(false);
   433→      setTimeout(() => setRedeemSuccess(null), 3000);
   434→    } catch (error: unknown) {
   435→      console.error("[Redeem] Error:", error);
   436→      const msg = error instanceof Error ? error.message : t("credit.redeemError");
   437→      toast.error(msg.includes("permission") ? t("credit.noPermission") : t("credit.redeemErrorRetry"));
   438→    }
   439→    setIsRedeeming(false);
   440→  };
   441→
   442→  const handleSubscribe = async (plan: SubscriptionPlan & { id: string }) => {
   443→    if (!user) return;
   444→    setIsSubscribing(true);
   445→    try {
   446→      // Use runTransaction on amount field only to avoid race conditions
   447→      const amountRef = ref(db, `credit/${user.uid}/amount`);
   448→      const result = await runTransaction(amountRef, (currentAmount) => {
   449→        const bal = (currentAmount || 0) as number;
   450→        if (bal < plan.price) return; // abort - insufficient balance
   451→        return bal - plan.price;
   452→      });
   453→
   454→      if (!result.committed) {
   455→        toast.error(`${t("credit.insufficientBalanceNeed")} ${plan.price.toLocaleString()} ${t("credit.riyalShort")}`);
   456→        setIsSubscribing(false);
   457→        return;
   458→      }
   459→
   460→      // Update updatedAt separately
   461→      await update(ref(db, `credit/${user.uid}`), { updatedAt: Date.now() });
   462→
   463→      const histRef = push(ref(db, `credit/${user.uid}/history`));
   464→      await set(histRef, { type: "purchase", amount: plan.price, description: `${t("credit.subscribePlanDesc")} ${plan.name}`, date: Date.now() });
   465→      await set(ref(db, `userSubscriptions/${user.uid}`), {
   466→        planId: plan.id,
   467→        planName: plan.name,
   468→        activatedAt: Date.now(),
   469→        expiresAt: Date.now() + plan.durationDays * 86400000,
   470→        isActive: true,
   471→        autoRenew: true,
   472→        uid: user.uid,
   473→      });
   474→      const notifRef = push(ref(db, `notifications/${user.uid}`));
   475→      await set(notifRef, {
