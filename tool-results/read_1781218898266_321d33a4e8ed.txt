    10→import { db } from "@/lib/firebase";

    40→  const { t, isRTL, lang } = useLanguage();
    41→
    42→  const [ads, setAds] = useState<Advertisement[]>([]);
    43→  const [homeBanners, setHomeBanners] = useState<HomeBanner[]>([]);
    44→  const [simCards, setSimCards] = useState<SimCard[]>([]);
    45→  const [currentAd, setCurrentAd] = useState(0);
    46→  const [currentBanner, setCurrentBanner] = useState(0);
    47→  const [fbNetworks, setFbNetworks] = useState<NetworkItem[]>(DEFAULT_NETWORKS.map(n => ({
    48→    ...n, ownerId: null, ownerName: null, ownerPhone: null, location: null,
    49→    provinceId: null, provinceName: null, district: null, exactLocation: null,
    50→    connectionIP: null, imageBase64: null, networkType: null, coverage: null, speed: null, createdAt: 0
    51→  })));
    52→  const [allCards, setAllCards] = useState<CardItem[]>([]);
    53→  const [balance, setBalance] = useState(0);
    54→  const [maxBalance, setMaxBalance] = useState(0);
    55→  const [appDownloadUrl, setAppDownloadUrl] = useState("");
    56→
    57→  const [selectedProvince, setSelectedProvince] = useState<string | null>(null);
    58→  const [selectedDistrict, setSelectedDistrict] = useState<string | null>(null);
    59→  const [userProvinceId, setUserProvinceId] = useState<string | null>(null);
    60→
    61→  const [refreshing, setRefreshing] = useState(false);
    62→  const [pullY, setPullY] = useState(0);
    63→  const [isPulling, setIsPulling] = useState(false);
    64→  const touchStartY = useRef(0);
    65→  const scrollRef = useRef<HTMLDivElement>(null);
    66→
    67→  const [selectedNetwork, setSelectedNetwork] = useState<NetworkItem | null>(null);
    68→  const [networkSearch, setNetworkSearch] = useState("");
    69→
    70→  useEffect(() => {
    71→    const unsubs: (() => void)[] = [];
    72→    const adsUnsub = onValue(ref(db, "advertisements"), (snap) => {
    73→      const data = snap.val();

    81→        setHomeBanners(banners);
    82→      } else setHomeBanners([]);
    83→    });

    85→    const simsUnsub = onValue(ref(db, "simCards"), (snap) => {
    86→      const data = snap.val();

    88→    });

    90→    const netUnsub = onValue(ref(db, "networks"), (snap) => {

   101→      unsubs.push(onValue(ref(db, `credit/${user.uid}/amount`), (snap) => setBalance(snap.val() || 0)));

   116→  }, [homeBanners.length]);
   117→

   119→    if (ads.length <= 1) return;
   120→    const timer = setInterval(() => setCurrentAd((prev) => (prev + 1) % ads.length), 5000);
   121→    return () => clearInterval(timer);
   122→  }, [ads.length]);
   123→

   127→    const available = getAvailableCount(net.id);
   128→    if (available === 0) return false;
   129→    if (selectedProvince && net.provinceId !== selectedProvince) return false;
   130→    if (selectedDistrict && net.district !== selectedDistrict) return false;
   131→    return true;
   132→  }).sort((a, b) => {
   133→    const aNear = a.provinceId === userProvinceId ? 1 : 0;
   134→    const bNear = b.provinceId === userProvinceId ? 1 : 0;
   135→    if (aNear !== bNear) return bNear - aNear;
   136→    return getAvailableCount(b.id) - getAvailableCount(a.id);
   137→  });
   138→
   139→  const filteredNetworks = fbNetworks.filter(net => {
   140→    if (selectedProvince && net.provinceId !== selectedProvince) return false;
   141→    if (selectedDistrict && net.district !== selectedDistrict) return false;
   142→    if (networkSearch && !net.name.toLowerCase().includes(networkSearch.toLowerCase())) return false;
   143→    return true;
   144→  });

   146→  const districts = selectedProvince ? getDistricts(selectedProvince) : [];
   147→  const districtsEn = selectedProvince ? getDistrictsEn(selectedProvince) : [];

   151→    if (el && el.scrollTop <= 0) { touchStartY.current = e.touches[0].clientY; setIsPulling(true); }
   152→  }, []);

   154→    if (!isPulling) return;
   155→    const el = scrollRef.current;
   156→    if (el && el.scrollTop > 0) { setIsPulling(false); setPullY(0); return; }

   158→    if (diff > 0) setPullY(Math.min(diff * 0.4, 100));

   164→    } else { setPullY(0); setIsPulling(false); }

   166→
   167→  const activeBanners = homeBanners.length > 0 ? homeBanners : ads.length > 0 ? null : null;
   168→

   172→      animate={{ opacity: 1, y: 0 }}
   173→      exit={{ opacity: 0, y: -16 }}
   174→      transition={{ type: "spring", stiffness: 120, damping: 14 }}
   175→      className="relative"

   178→      <motion.div
   179→        className="flex items-center justify-center py-2 overflow-hidden"
   180→        animate={{ height: pullY > 0 || refreshing ? (refreshing ? 44 : pullY * 0.5) : 0 }}
   181→        transition={{ type: "spring", stiffness: 300, damping: 25 }}
   182→      >
   183→        <motion.div animate={{ rotate: refreshing ? 360 : 0 }} transition={{ repeat: refreshing ? Infinity : 0, duration: 0.8, ease: "linear" }}>
   184→          <RefreshCw className={`w-5 h-5 ${refreshing ? "text-[#1B7A3D]" : "text-gray-300 dark:text-slate-600"}`} />
   185→        </motion.div>
   186→        {pullY > 30 && !refreshing && (
   187→          <span className="text-[10px] text-gray-400 dark:text-slate-500 mx-2">
   188→            <ArrowDown className="w-3 h-3 inline" /> {t("home.pullToRefresh")}
   189→          </span>
   190→        )}
   191→        {refreshing && <span className="text-[10px] text-[#1B7A3D] mx-2 font-bold">{t("home.refreshing")}</span>}
   192→      </motion.div>
   193→
   194→      <div
   195→        ref={scrollRef}
   196→        onTouchStart={handleTouchStart}
   197→        onTouchMove={handleTouchMove}
   198→        onTouchEnd={handleTouchEnd}
   199→        className="px-4 pt-3 pb-6 space-y-4"
   200→      >
   201→
   202→        {/* ══ 1. HERO BALANCE CARD ══ */}
   203→        <motion.div
   204→          initial={{ scale: 0.97, opacity: 0 }}
   205→          animate={{ scale: 1, opacity: 1 }}
   206→          transition={{ type: "spring", stiffness: 200, damping: 18, delay: 0.05 }}
   207→          className="relative overflow-hidden rounded-3xl"
   208→        >
   209→          <div className="bg-gradient-to-br from-[#0d5c2e] via-[#1B7A3D] to-[#22A24D] rounded-3xl p-5 relative overflow-hidden">
   210→            {/* Decorative blobs */}
   211→            <div className="absolute -top-8 -right-8 w-32 h-32 bg-white/10 rounded-full blur-xl" />
   212→            <div className="absolute -bottom-8 -left-8 w-40 h-40 bg-white/5 rounded-full blur-2xl" />
   213→            <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 w-56 h-56 bg-white/[0.03] rounded-full" />
   214→
   215→            <div className="relative z-10">
   216→              {/* App branding row */}
   217→              <div className="flex items-center justify-between mb-4">
   218→                <div className="flex items-center gap-2.5">
   219→                  <div className="w-10 h-10 rounded-2xl overflow-hidden ring-2 ring-white/30 shadow-lg">
   220→                    <img src="/images/IMG_20260527_220851.jpg" alt="Apple.NET" className="w-full h-full object-cover" />
   221→                  </div>
   222→                  <div>
   223→                    <p className="text-white/50 text-[10px] font-medium tracking-wider uppercase">Apple.NET</p>
   224→                    <h1 className="text-base font-black text-white leading-tight">
   225→                      {user ? t("home.welcomeBack") : t("home.welcome")}
   226→                    </h1>
   227→                  </div>
   228→                </div>
   229→                {!user && (
   230→                  <Button
   231→                    onClick={onAuthClick}
   232→                    size="sm"
   233→                    className="bg-white/20 hover:bg-white/30 text-white border border-white/25 font-bold rounded-xl text-xs h-8 px-3 backdrop-blur-sm"
   234→                  >
   235→                    {t("home.signIn")}
   236→                  </Button>
   237→                )}
   238→              </div>
   239→
   240→              {user ? (
   241→                <>
   242→                  {/* Balance display */}
   243→                  <div className="bg-white/10 rounded-2xl px-4 py-3.5 mb-4 backdrop-blur-sm border border-white/10">
   244→                    <p className="text-white/50 text-[10px] font-medium mb-1">{t("home.currentBalance")}</p>
   245→                    <motion.p
   246→                      key={balance}
   247→                      initial={{ scale: 0.85, opacity: 0 }}
   248→                      animate={{ scale: 1, opacity: 1 }}
   249→                      transition={{ type: "spring", stiffness: 300, damping: 20 }}
   250→                      className="text-3xl font-black text-white tracking-tight"

   252→                      {balance.toLocaleString()} <span className="text-base font-medium text-white/60">{t("home.yer")}</span>
   253→                    </motion.p>

   259→                        </div>
   260→                        <div className="h-1 bg-white/15 rounded-full overflow-hidden">
   261→                          <div className="h-full bg-white/50 rounded-full transition-all duration-700" style={{ width: `${Math.min((balance / maxBalance) * 100, 100)}%` }} />
   262→                        </div>
   263→                      </div>
   264→                    )}
   265→                  </div>
   266→
   267→                  {/* Quick actions */}
   268→                  <div className="grid grid-cols-2 gap-2">
   269→                    <Button
   270→                      onClick={() => onNavigate("cards")}
   271→                      className="bg-white text-[#1B7A3D] font-black rounded-xl h-11 text-sm hover:bg-white/90 shadow-lg haptic-press"
   272→                    >
   273→                      <ShoppingBag className="w-4 h-4 mx-1.5" />
   274→                      {t("home.buyCards")}
   275→                    </Button>
   276→                    <Button
   277→                      onClick={() => onNavigate("deposit")}
   278→                      className="bg-white/15 hover:bg-white/25 text-white border border-white/25 font-bold rounded-xl h-11 text-sm haptic-press"
   279→                    >
   280→                      <Wallet className="w-4 h-4 mx-1.5" />
   281→                      {t("home.deposit")}
   282→                    </Button>

   286→                <>
   287→                  <p className="text-white/65 text-sm leading-relaxed text-center mb-4 max-w-xs mx-auto">
   288→                    {t("home.platform")}
   289→                  </p>
   290→                  <div className="grid grid-cols-2 gap-2">
   291→                    <Button onClick={onAuthClick} className="bg-white text-[#1B7A3D] font-black rounded-xl h-11 text-sm hover:bg-white/90 shadow-lg haptic-press">
   292→                      {t("home.signIn")}
   293→                    </Button>
   294→                    <Button onClick={() => onNavigate("cards")} className="bg-white/15 hover:bg-white/25 text-white border border-white/25 font-bold rounded-xl h-11 text-sm haptic-press">
   295→                      <Wifi className="w-4 h-4 mx-1" />{t("home.browseCards")}
   296→                    </Button>
   297→                  </div>
   298→                </>
   299→              )}
   300→            </div>
   301→          </div>
   302→        </motion.div>
   303→
   304→        {/* ══ 2. QUICK SERVICES ══ */}
   305→        <div>
   306→          <p className="text-[11px] font-bold text-gray-400 dark:text-slate-500 mb-2 px-1 uppercase tracking-wider">
   307→            {t("home.services")}
   308→          </p>
   309→          <div className="grid grid-cols-4 gap-2">
   310→            {[
   311→              { icon: Wifi, label: t("home.cards"), color: "bg-[#E8F5E9] dark:bg-green-900/30 text-[#1B7A3D]", action: () => onNavigate("cards") },
   312→              { icon: CreditCard, label: t("home.deposit"), color: "bg-blue-50 dark:bg-blue-900/30 text-blue-500 dark:text-blue-400", action: () => onNavigate("deposit") },
   313→              { icon: Satellite, label: "Starlink", color: "bg-purple-50 dark:bg-purple-900/30 text-purple-500 dark:text-purple-400", action: () => onNavigate("starlink") },
   314→              { icon: Smartphone, label: t("home.telecom"), color: "bg-orange-50 dark:bg-orange-900/30 text-orange-500 dark:text-orange-400", action: () => onNavigate("telecom") },
   315→            ].map((item, i) => (
   316→              <motion.div
   317→                key={i}
   318→                whileTap={{ scale: 0.9 }}
   319→                className="bg-white dark:bg-slate-800 rounded-2xl p-3 text-center card-shadow cursor-pointer haptic-press border border-gray-100/50 dark:border-slate-700/50"
   320→                onClick={item.action}

   328→          </div>

   331→        {/* ══ 3. BANNER / ADS ══ */}

   334→            <AnimatePresence mode="wait">
   335→              <motion.div
   336→                key={homeBanners.length > 0 ? currentBanner : currentAd}
   337→                initial={{ opacity: 0, x: isRTL ? 40 : -40 }}
   338→                animate={{ opacity: 1, x: 0 }}
   339→                exit={{ opacity: 0, x: isRTL ? -40 : 40 }}
   340→                transition={{ duration: 0.4 }}
   341→                className="relative aspect-[16/7] bg-slate-100 dark:bg-slate-800"
   342→              >
   343→                <img

   345→                    ? (homeBanners[currentBanner]?.imageUrl || "/images/IMG-20260527-WA0043.jpg")
   346→                    : (ads[currentAd]?.imageUrl || "/images/IMG-20260527-WA0043.jpg")
   347→                  }
   348→                  alt="banner"
   349→                  className="w-full h-full object-cover"
   350→                />
   351→                <div className="absolute inset-0 bg-gradient-to-t from-black/70 via-black/10 to-transparent" />

   353→                  <Badge className="bg-[#1B7A3D] text-white text-[8px] mb-1 gap-1">
   354→                    <Megaphone className="w-2.5 h-2.5" />{t("home.ad")}
   355→                  </Badge>
   356→                  <h3 className="text-sm font-bold text-white">
   357→                    {homeBanners.length > 0 ? homeBanners[currentBanner]?.title : ads[currentAd]?.title}
   358→                  </h3>
   359→                </div>
   360→              </motion.div>
   361→            </AnimatePresence>

   363→            {(homeBanners.length > 1 || ads.length > 1) && (

   371→                ))}
   372→              </div>
   373→            )}
   374→          </div>

   377→        {/* ══ 4. STATS ROW ══ */}

   379→          {[
   380→            { value: "99.9%", label: t("home.successRate"), icon: TrendingUp, color: "text-emerald-500" },
   381→            { value: "10K+", label: t("home.activeUsers"), icon: Star, color: "text-amber-500" },
   382→            { value: String(vendingNetworks.length || fbNetworks.length), label: t("home.networks"), icon: Zap, color: "text-blue-500" },
   383→          ].map((stat, i) => (

   385→              key={i}
   386→              initial={{ opacity: 0, y: 8 }}
   387→              animate={{ opacity: 1, y: 0 }}
   388→              transition={{ delay: 0.1 + i * 0.05 }}
   389→              className="bg-white dark:bg-slate-800 rounded-2xl p-3 text-center card-shadow border border-gray-100/50 dark:border-slate-700/50"

   391→              <stat.icon className={`w-4 h-4 mx-auto mb-1 ${stat.color}`} />

   399→        <div>
   400→          <div className="flex items-center justify-between mb-3">
   401→            <div className="flex items-center gap-2">
   402→              <div className="w-7 h-7 rounded-xl bg-[#E8F5E9] dark:bg-green-900/30 flex items-center justify-center">
   403→                <Package className="w-3.5 h-3.5 text-[#1B7A3D]" />
   404→              </div>
   405→              <div>
   406→                <h2 className="text-sm font-black text-gray-900 dark:text-white">
   407→                  {t("home.availableMachines")}
   408→                </h2>
   409→                <p className="text-[10px] text-gray-400 dark:text-slate-500">

   413→            </div>
